┌─────────────────────────────────────────────────────────────────┐
│                    KKYX Spider 配置快速参考                      │
└─────────────────────────────────────────────────────────────────┘

┌─ 配置文件层次 (优先级从高到低) ───────────────────────────────────┐
│ 1. .config.py          ← 本地覆盖配置 (可选)                    │
│ 2. config.py           ← 默认配置 (主文件)                       │
│ 3. .env                ← 敏感凭据 (环境变量)                     │
└───────────────────────────────────────────────────────────────────┘

┌─ 必需的 .env 参数 ───────────────────────────────────────────────┐
│ KKYX_USER=your_username        # KKYX 网站用户名                 │
│ KKYX_PWD=your_password         # KKYX 网站密码                   │
└───────────────────────────────────────────────────────────────────┘

┌─ 可选的 .env 参数 ───────────────────────────────────────────────┐
│ WP_BASE_URL=https://...        # WordPress 站点 URL              │
│ WP_USERNAME=admin             # WordPress 用户名                │
│ WP_APP_PASSWORD="..."         # WordPress 应用密码              │
│ CDN_BASE_URL=https://...      # CDN 基础 URL                     │
└───────────────────────────────────────────────────────────────────┘

┌─ .config.py 示例 ────────────────────────────────────────────────┐
│ config_overrides = {                                                │
│     "safety": {                                                     │
│         "index_scan_page_limit": 50,   # 每次扫描页面数           │
│         "max_posts_per_run": 20,        # 每次处理帖子数          │
│     },                                                              │
│     "debug": {                                                      │
│         "debug_mode": False,            # 关闭调试模式            │
│         "log_level": "INFO",            # 日志级别                │
│     },                                                              │
│ }                                                                    │
└───────────────────────────────────────────────────────────────────┘

┌─ 重要目录 (均在 .gitignore 中) ───────────────────────────────────┐
│ .data/storage/     ← 图片存储                                    │
│ .data/db/          ← 数据库文件                                  │
│ .screenshot/       ← 截图存储                                    │
│ .auth/             ← 登录会话                                    │
└───────────────────────────────────────────────────────────────────┘

┌─ 常用命令 ────────────────────────────────────────────────────────┐
│ # 验证配置加载                                                     │
│ python3 -c "import config; print(config.MAX_POSTS_PER_RUN)"       │
│                                                                     │
│ # 首次运行 (自动登录)                                              │
│ python3 .auth/auto_login.py                                       │
│                                                                     │
│ # 运行爬虫                                                         │
│ uv run main.py                                                     │
│                                                                     │
│ # 启动 Web 查看器                                                  │
│ python3 web/app.py                                                 │
└───────────────────────────────────────────────────────────────────┘

┌─ 文件权限 ────────────────────────────────────────────────────────┐
│ chmod 600 .env          # 设置敏感文件权限                        │
│ chmod 600 .config.py    # 设置配置文件权限                        │
└───────────────────────────────────────────────────────────────────┘

┌─ 故障排除 ────────────────────────────────────────────────────────┐
│ • 配置未生效？检查 .config.py 中字典结构是否正确                  │
│ • 找不到模块？确保在项目根目录运行                                 │
│ • 环境变量未生效？检查 .env 文件格式和引号                         │
└───────────────────────────────────────────────────────────────────┘
